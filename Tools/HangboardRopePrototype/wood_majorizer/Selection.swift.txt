extension RopeTriangleCollider {
    static func majorizerWoodSelection(_ start:SIMD3<Double>,_ end:SIMD3<Double>,_ triangle:[SIMD3<Double>]) throws -> WoodMajorizer.Selection {
        guard triangle.count==3,(triangle+[start,end]).allSatisfy({[$0.x,$0.y,$0.z].allSatisfy{$0.isFinite}}),
              simd_length_squared(simd_cross(triangle[1]-triangle[0],triangle[2]-triangle[0]))>0 else {
            throw RopePhysicsError.invalid("Invalid direct wood triangle")
        }
        let a=triangle[0],b=triangle[1],c=triangle[2]
        func region(_ p:SIMD3<Double>)->(String,[String]) {
            let ab=b-a,ac=c-a,ap=p-a,d1=simd_dot(ab,ap),d2=simd_dot(ac,ap)
            let bp=p-b,d3=simd_dot(ab,bp),d4=simd_dot(ac,bp)
            let vc=d1*d4-d3*d2
            let cp=p-c,d5=simd_dot(ab,cp),d6=simd_dot(ac,cp)
            let vb=d5*d2-d1*d6
            let va=d3*d6-d5*d4
            let predicates:[(String,Double)]=[("d1",d1),("d2",d2),("d3",d3),("d4-d3",d4-d3),("d6",d6),("d5-d6",d5-d6),("va",va),("vb",vb),("vc",vc)]
            let boundaries=predicates.filter{$0.1==0}.map{"exactPredicate:\($0.0)"}
            if d1<=0 && d2<=0 {return ("vertexA",boundaries)}
            if d3>=0 && d4<=d3 {return ("vertexB",boundaries)}
            if vc<=0 && d1>=0 && d3<=0 {return ("edgeAB",boundaries)}
            if d6>=0 && d5<=d6 {return ("vertexC",boundaries)}
            if vb<=0 && d2>=0 && d6<=0 {return ("edgeAC",boundaries)}
            if va<=0 && d4-d3>=0 && d5-d6>=0 {return ("edgeBC",boundaries)}
            return ("face",boundaries)
        }
        func pairBranch(_ p:SIMD3<Double>,_ q:SIMD3<Double>,_ a:SIMD3<Double>,_ b:SIMD3<Double>)->WoodMajorizer.PairBranch {
            let d1=q-p,d2=b-a,r=p-a,aa=simd_dot(d1,d1),ee=simd_dot(d2,d2),f=simd_dot(d2,r)
            func clamp(_ value:Double)->String {value<=0 ? "zero":(value>=1 ? "one":"free")}
            if aa<=1e-24 {
                let t=f/ee,label=clamp(t)
                return .init(ropeFormula:"degenerate",ropeClamp:"zero",triangleFormula:label=="free" ? "degenerateRopeProjection":label,boundaries:(t==0 || t==1 ? ["triangleClampBoundary"]:[])+(aa==1e-24 ? ["ropeDegeneracyBoundary"]:[]))
            }
            let c=simd_dot(d1,r)
            if ee<=1e-24 {let s = -c/aa;return .init(ropeFormula:"triangleStart",ropeClamp:clamp(s),triangleFormula:"zero",boundaries:(s==0 || s==1 ? ["ropeClampBoundary"]:[])+(ee==1e-24 ? ["triangleDegeneracyBoundary"]:[]))}
            let bb=simd_dot(d1,d2),denominator=aa*ee-bb*bb
            let rawS=denominator>1e-24 ? (bb*f-c*ee)/denominator:0,s=min(1,max(0,rawS)),t=(bb*s+f)/ee
            var boundaries:[String]=[]
            if rawS==0 || rawS==1 {boundaries.append("ropeClampBoundary")}
            if t==0 || t==1 {boundaries.append("triangleClampBoundary")}
            if denominator<=1e-24 {boundaries.append("parallelBranch")}
            if denominator==1e-24 {boundaries.append("parallelPredicateBoundary")}
            if t<0 {let value = -c/aa;if value==0 || value==1 {boundaries.append("ropeFinalClampBoundary")};return .init(ropeFormula:"triangleStart",ropeClamp:clamp(value),triangleFormula:"zero",boundaries:boundaries)}
            if t>1 {let value=(bb-c)/aa;if value==0 || value==1 {boundaries.append("ropeFinalClampBoundary")};return .init(ropeFormula:"triangleEnd",ropeClamp:clamp(value),triangleFormula:"one",boundaries:boundaries)}
            return .init(ropeFormula:denominator>1e-24 ? "interior":"parallelZero",ropeClamp:clamp(rawS),triangleFormula:"projected",boundaries:boundaries)
        }
        let firstRegion=region(start)
        var candidates=[WoodMajorizer.Candidate(name:"start",point:start,surface:Self.triangleClosest(start,a,b,c),fraction:0,region:firstRegion.0,pair:nil,boundaries:firstRegion.1)]
        if start != end {
            let lastRegion=region(end)
            candidates.append(.init(name:"end",point:end,surface:Self.triangleClosest(end,a,b,c),fraction:1,region:lastRegion.0,pair:nil,boundaries:lastRegion.1))
            if let t=Self.rayTriangle(start,end-start,a,b,c),t>=0,t<=1 {
                let p=start+(end-start)*t;candidates.append(.init(name:"ray",point:p,surface:p,fraction:t,region:"intersection",pair:nil))
            }
            for edge in 0..<3 {
                let u=triangle[edge],v=triangle[(edge+1)%3],pair=Self.segmentPair(start,end,u,v),branch=pairBranch(start,end,u,v)
                candidates.append(.init(name:"edge\(edge)",point:pair.0,surface:pair.1,fraction:pair.2,region:"segmentPair",pair:branch,boundaries:branch.boundaries))
            }
        }
        var winner=0,best=simd_length_squared(candidates[0].point-candidates[0].surface)
        for i in candidates.indices where i>0 {
            let square=simd_length_squared(candidates[i].point-candidates[i].surface)
            if square<best {winner=i;best=square}
        }
        guard best.isFinite,sqrt(best)>1e-10 else {throw RopePhysicsError.invalid("Degenerate or intersecting direct wood feature")}
        let ties=candidates.indices.filter{$0 != winner && simd_length_squared(candidates[$0].point-candidates[$0].surface)==best}
        return .init(start:start,end:end,triangle:triangle,candidates:candidates,winner:winner,ties:ties)
    }
}
