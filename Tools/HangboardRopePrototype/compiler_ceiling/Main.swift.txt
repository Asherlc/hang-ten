import Foundation
import CryptoKit
import simd
import Darwin
let root=URL(fileURLWithPath:CommandLine.arguments[1])
let label=CommandLine.arguments[3]
let data=try Data(contentsOf:URL(fileURLWithPath:"Hangboards/clavellium-training-block/assets/primary.physics.json"))
let raw=try JSONSerialization.jsonObject(with:data) as! [String:Any]
let input=try RopePhysicsDescriptor.decode(data).validated(modelSHA256:raw["modelSHA256"] as! String)
let collider=try RopeTriangleCollider(input:input)
let checkpoint=try Data(contentsOf:URL(fileURLWithPath:CommandLine.arguments[2]))
var candidate=try RopeDynamicsSolver.restoreBundle(input:input,collider:collider,data:checkpoint,strict:false)
candidate.woodMajorizerExperiment=true;candidate.woodResidualExperiment=true;candidate.woodFeatureIdentityExperiment=true
candidate.preconditionedResidualExperiment=true;candidate.stationaryResidualExperiment=true
SolverCollection.collect=false;ResidualStopTrace.enabled=true
let target=simd_quatd(angle:Double.pi/6,axis:SIMD3<Double>(0,0,1))
var records:[[String:Any]]=[]
func save()throws {try JSONSerialization.data(withJSONObject:records,options:[.sortedKeys,.prettyPrinted]).write(to:root.appendingPathComponent(label+".json"))}
do {
 for i in 1...20 {
  alarm(10)
  let start=DispatchTime.now().uptimeNanoseconds
  let frame=try candidate.step(dt:1.0/240,targetOrientation:target)
  let seconds=Double(DispatchTime.now().uptimeNanoseconds-start)/1e9
  alarm(0)
  let bytes=try JSONSerialization.data(withJSONObject:candidate.bundleCheckpoint(),options:.sortedKeys)
  let digest=SHA256.hash(data:bytes).map{String(format:"%02x",$0)}.joined()
  let physical=try RopeSimulationMetrics.measure(state:candidate.state,input:input,collider:collider,
   boardHistory:(candidate.foundationCheckpoint()["history"] as! [[Double]]).map{$0[1]})
  records.append(["step":i,"seconds":seconds,"checkpointSHA256":digest,
   "QPs":candidate.reviewStepCorrections,"stationaryAccepted":candidate.reviewStationaryStepAccepted,
   "caps":candidate.reviewStepCaps,"retries":candidate.reviewStepRetries,"settled":frame.settled,
   "strain":physical.maximumLocalStrain,"materialError":physical.totalLengthError,
   "clearanceMargin":physical.minimumClearanceMargin,"geometryAccepted":physical.geometryAccepted])
  try save()
  guard physical.geometryAccepted,frame.metrics.geometryAccepted,candidate.reviewStepCaps==0,candidate.reviewStepRetries==0 else {
   throw RopePhysicsError.invalid("compiler corpus physical/cap/retry")
  }
 }
 print("PASS",label,records.reduce(0){$0+($1["seconds"] as! Double)})
} catch {try save();print("FAIL",label,error);exit(3)}
