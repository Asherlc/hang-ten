402: private func gripHandDiagnosticFieldOfView(points: [SIMD4<Float>],
403:                                            cameraTransform: simd_float4x4,
404:                                            aspect: Float) -> Float? {
405:     guard !points.isEmpty, aspect.isFinite, aspect > 0 else { return nil }
406:     let inverse = simd_inverse(cameraTransform)
407:     var tangent: Float = 0
408:     for point in points {
409:         let projected = inverse * point
410:         let depth = -projected.z
411:         guard projected.x.isFinite, projected.y.isFinite, depth.isFinite,
412:               depth > 0.1, depth < 100 else { return nil }
413:         tangent = max(tangent, abs(projected.y) / depth,
414:                       abs(projected.x) / (aspect * depth))
415:     }
416:     tangent *= 1.08
417:     let degrees = 2 * atan(tangent) * 180 / Float.pi
418:     guard tangent.isFinite, tangent > 0, degrees.isFinite,
419:           degrees > 0, degrees < 180 else { return nil }
420:     return degrees
421: }

589:     func orbit(azimuthDelta: Float, elevationDelta: Float, zoomScale: Float = 1) {
590:         guard azimuthDelta.isFinite, elevationDelta.isFinite,
591:               zoomScale.isFinite, zoomScale > 0 else { return }
592:         let fullRotation: Float = .pi * 2
593:         let nextAzimuth = (orbitAzimuth + azimuthDelta).truncatingRemainder(dividingBy: fullRotation)
594:         let nextElevation = min(max(orbitElevation + elevationDelta, -0.55), 0.55)
595:         let nextZoom = min(max(orbitZoom * zoomScale, 0.75), 1.35)
596:         let baseDistance = simd_length(canonicalOffset)
597:         guard baseDistance > 1e-6 else { return }
598:         let baseDirection = canonicalOffset / baseDistance
599:         let worldUp = SIMD3<Float>(0, 1, 0)
600:         let rightVector = simd_cross(worldUp, baseDirection)
601:         guard simd_length(rightVector) > 1e-6 else { return }
602:         let right = simd_normalize(rightVector)
603: 
604:         let yaw = simd_quatf(angle: nextAzimuth, axis: worldUp)
605:         let pitch = simd_quatf(angle: nextElevation, axis: right)
606:         let distance = baseDistance / nextZoom
607:         guard distance.isFinite, distance > 0 else { return }
608:         let rotatedOffset = (pitch * yaw).act(baseDirection) * distance
609:         guard rotatedOffset.x.isFinite, rotatedOffset.y.isFinite, rotatedOffset.z.isFinite,
610:               let transform = Self.cameraTransform(position: canonicalCenter + rotatedOffset,
611:                                                    target: canonicalCenter) else { return }
612:         camera.transform = Transform(matrix: transform)
613:         #if DEBUG
614:         if !BoardHighlightDiagnostic.usesPerspectiveHandCameras {
615:             var component = camera.components[OrthographicCameraComponent.self]!
616:             component.scale = canonicalOrthographicScale / nextZoom
617:             camera.components.set(component)
618:         }
619:         #else
620:         var component = camera.components[OrthographicCameraComponent.self]!
621:         component.scale = canonicalOrthographicScale / nextZoom
622:         camera.components.set(component)
623:         #endif
624:         orbitAzimuth = nextAzimuth
625:         orbitElevation = nextElevation
626:         orbitZoom = nextZoom
627:     }
