473:     func resetCamera() {
474:         guard let surface else { return }
475:         let points = surface.posedVerticesForFraming().map { point in
476:             SIMD4<Float>(point.x * hand.scale.x, point.y, point.z, 1)
477:         }
478:         guard !points.isEmpty else { return }
479:         var low = SIMD3<Float>(repeating: .greatestFiniteMagnitude)
480:         var high = SIMD3<Float>(repeating: -.greatestFiniteMagnitude)
481:         for point in points {
482:             let xyz = SIMD3<Float>(point.x, point.y, point.z)
483:             low = simd_min(low, xyz)
484:             high = simd_max(high, xyz)
485:         }
486:         let center = (low + high) / 2
487:         let offset = SIMD3<Float>(currentSide == .left ? 5.8 : -5.8, 2.75, 9.4)
488:         guard let transform = Self.cameraTransform(position: center + offset, target: center) else { return }
489:         let inverseCamera = simd_inverse(transform)
490:         var halfWidth: Float = 0
491:         var halfHeight: Float = 0
492:         for point in points {
493:             let projected = inverseCamera * point
494:             halfWidth = max(halfWidth, abs(projected.x))
495:             halfHeight = max(halfHeight, abs(projected.y))
496:         }
497:         let size = currentViewportSize
498:         let aspect = size.width.isFinite && size.height.isFinite && size.width > 0 && size.height > 0
499:             ? Float(size.width / size.height) : 0.85
500:         guard aspect.isFinite, aspect > 0 else { return }
501:         let orthographicScale = max(halfHeight, halfWidth / aspect) * 1.08
502:         guard orthographicScale.isFinite, orthographicScale > 0 else { return }
503:         camera.transform = Transform(matrix: transform)
504:         var component = camera.components[OrthographicCameraComponent.self]!
505:         component.scale = orthographicScale

516:     func orbit(azimuthDelta: Float, elevationDelta: Float, zoomScale: Float = 1) {
517:         guard azimuthDelta.isFinite, elevationDelta.isFinite,
518:               zoomScale.isFinite, zoomScale > 0 else { return }
519:         let fullRotation: Float = .pi * 2
520:         let nextAzimuth = (orbitAzimuth + azimuthDelta).truncatingRemainder(dividingBy: fullRotation)
521:         let nextElevation = min(max(orbitElevation + elevationDelta, -0.55), 0.55)
522:         let nextZoom = min(max(orbitZoom * zoomScale, 0.75), 1.35)
523:         let baseDistance = simd_length(canonicalOffset)
524:         guard baseDistance > 1e-6 else { return }
525:         let baseDirection = canonicalOffset / baseDistance
526:         let worldUp = SIMD3<Float>(0, 1, 0)
527:         let rightVector = simd_cross(worldUp, baseDirection)
528:         guard simd_length(rightVector) > 1e-6 else { return }
529:         let right = simd_normalize(rightVector)
530: 
531:         let yaw = simd_quatf(angle: nextAzimuth, axis: worldUp)
532:         let pitch = simd_quatf(angle: nextElevation, axis: right)
533:         let distance = baseDistance / nextZoom
534:         guard distance.isFinite, distance > 0 else { return }
535:         let rotatedOffset = (pitch * yaw).act(baseDirection) * distance
536:         guard rotatedOffset.x.isFinite, rotatedOffset.y.isFinite, rotatedOffset.z.isFinite,
537:               let transform = Self.cameraTransform(position: canonicalCenter + rotatedOffset,
538:                                                    target: canonicalCenter) else { return }
539:         camera.transform = Transform(matrix: transform)
540:         var component = camera.components[OrthographicCameraComponent.self]!
541:         component.scale = canonicalOrthographicScale / nextZoom
542:         camera.components.set(component)
543:         orbitAzimuth = nextAzimuth
544:         orbitElevation = nextElevation
545:         orbitZoom = nextZoom
546:     }

713:     func orbit(azimuthDelta: Float, elevationDelta: Float, zoomScale: Float = 1) {
714:         guard azimuthDelta.isFinite, elevationDelta.isFinite,
715:               zoomScale.isFinite, zoomScale > 0 else { return }
716:         let nextAzimuth = (orbitAzimuth + azimuthDelta).truncatingRemainder(dividingBy: .pi * 2)
717:         let nextElevation = min(max(orbitElevation + elevationDelta, -0.55), 0.55)
718:         let nextZoom = min(max(orbitZoom * zoomScale, 0.75), 1.35)
719:         let distance = simd_length(canonicalOffset)
720:         guard distance.isFinite, distance > 1e-6 else { return }
721:         let direction = canonicalOffset / distance
722:         let up = SIMD3<Float>(0, 1, 0)
723:         let rightVector = simd_cross(up, direction)
724:         guard simd_length(rightVector) > 1e-6 else { return }
725:         let yaw = simd_quatf(angle: nextAzimuth, axis: up)
726:         let pitch = simd_quatf(angle: nextElevation, axis: simd_normalize(rightVector))
727:         let rotatedOffset = (pitch * yaw).act(direction) * (distance / nextZoom)
728:         guard let transform = Self.cameraTransform(position: canonicalCenter + rotatedOffset,

736:         orbitZoom = nextZoom
