1342:         board: BoardRevision,
1343:         presentation: BoardPresentation,
1344:         store: BoardPackageStore = BoardCatalog.packageStore,

1348:             throw BoardModelRealityError.presentationNotModel

1350:

1356:         let key = BoardModelRealityKey(
1357:             boardID: board.id,

1359:             modelSHA256: media.descriptor.modelSHA256

1362:         guard let source = try await BoardModelRealityCache.source(
1363:             for: key,
1364:             media: media,
1365:             board: board,
1366:             presentationID: presentation.id,
1367:             store: store,
1368:             resourceAccess: resourceAccess
1369:         ), !Task.isCancelled else {
1370:             if Task.isCancelled { throw CancellationError() }

1376:             display: media.display,
1377:             suspension: media.suspension,
1378:             orientation: media.orientation,

1380:                 $0.presentationID == presentation.id

1384:         )
1385:
