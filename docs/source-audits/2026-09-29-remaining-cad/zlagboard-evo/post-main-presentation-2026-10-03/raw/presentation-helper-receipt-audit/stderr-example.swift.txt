// Prospective DEBUG helper snippet only; not applied or compiled. Existing UIKit/Foundation imports suffice.
#if DEBUG
private func receipt(_ event: String, fields: String) {
    let line = "[BoardSimulatorPresentation] event=\(event) helper=\(ObjectIdentifier(self)) uptime=\(ProcessInfo.processInfo.systemUptime) \(fields)\n"
    do {
        try FileHandle.standardError.write(contentsOf: Data(line.utf8))
    } catch {
        // Diagnostic write failure must not change matching, retry or renderer behavior.
    }
}
#endif
